# OpenblH743VIT6 — custom STM32H743VIT6 flight controller board.
# 更新软件包列表并升级系统
sudo apt update && sudo apt upgrade

# 安装基础编译工具
# 你需要安装git（用于克隆代码）、make（用于构建）、clang（部分测试需要）和python等。
sudo apt -y install build-essential git curl clang-18 python3 python-is-python3

# 克隆Betaflight源代码
git clone https://github.com/betaflight/betaflight.git

# 进入源代码目录
cd betaflight

# 切换到稳定版本（例如4.5.3）
git checkout 4.5.3

# 先应用本地 SPL06 气压计补丁（在 Betafight 目录下）
cp -r <repo>/firmware固件/src编译源代码资料/Betafight/patches/drivers/barometer/* src/main/drivers/barometer/
cp -r <repo>/firmware固件/src编译源代码资料/Betafight/patches/drivers/compass/*   src/main/drivers/compass/
cp -r <repo>/firmware固件/src编译源代码资料/Betafight/patches/sensors/*           src/main/sensors/
cp    <repo>/firmware固件/src编译源代码资料/Betafight/patches/cli/settings.c      src/main/cli/

# 把 OpenblH743VIT6 文件夹放到 src/config/configs 目录下
# 4.5.3 不要执行 make configs（会去 clone GitHub 的 betaflight/config 仓库）

# 用已有的 GCC 10.x 工具链编译（官方 4.5.3 要求 10.3.1，覆盖版本检查）
make OpenblH743VIT6 ARM_SDK_DIR=$HOME/gcc-arm-none-eabi-10-2020-q4-major GCC_REQUIRED_VERSION=10.2.1 -j32

# 在 ~/betaflight/obj 查找固件 betaflight_4.5.3_STM32H743_OpenblH743VIT6.hex

# ===== 关键说明（详见 OpenblH743VIT6/README.md 和 patches/README.md）=====
# - HSE 晶振 8MHz：config.h 里 SYSTEM_HSE_MHZ 8
# - IMU 方向：CW90_DEG_FLIP
# - SPL06 气压计：已通过本地补丁支持，config.h 已启用 USE_BARO / USE_BARO_SPL06
# - QMC5883P 磁力计：已通过本地补丁支持，config.h 已启用 USE_MAG / USE_MAG_QMC5883P
# - SD 卡：暂不支持，USE_SDCARD 仍保持注释
# - 烧录后 CLI 执行 defaults + save（否则 IMU 方向读到旧配置值）
